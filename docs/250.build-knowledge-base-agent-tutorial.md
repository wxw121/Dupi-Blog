## 它解决什么问题

### 手动管理知识库的痛点

```text
传统方式：
  1. 运营人员手动上传文档
  2. 手动触发索引
  3. 文档更新后忘记重新索引
  4. 用户问到了过期内容
  5. 删除文档后向量没清理
  6. 不知道哪些文档索引失败了

后果：
  - 知识库内容过期
  - 检索结果不准确
  - 存储浪费（孤立向量）
  - 运维成本高
```

### Agent 自动化管理

```text
知识库 Agent 的职责：
  1. 监控文档变化（新增、修改、删除）
  2. 自动触发解析和索引
  3. 检测索引失败并自动重试
  4. 清理孤立向量
  5. 定期健康检查
  6. 回答"知识库状态"相关问题

价值：
  - 文档更新后 5 分钟内自动生效
  - 索引失败自动恢复
  - 零人工干预
  - 可观测、可追溯
```

## 核心概念：知识库 Agent 的职责

### 工具设计

```text
知识库 Agent 需要的工具：

文档管理工具：
  list_documents     列出所有文档及状态
  get_document       获取文档详情
  upload_document    上传新文档
  delete_document    删除文档（含向量清理）
  update_document    更新文档内容

索引管理工具：
  trigger_index      触发文档索引
  get_index_status   查询索引状态
  rebuild_index      重建整个索引
  cleanup_orphans    清理孤立向量

健康检查工具：
  check_health       综合健康检查
  get_stats          获取统计数据
  find_stale_docs    找出过期文档

查询工具：
  search_knowledge   语义搜索
  get_usage_stats    使用统计
```

### 文档生命周期状态机

```text
文档状态流转：

  uploaded → parsing → parsed → chunking → chunked → indexing → indexed
     ↓          ↓                   ↓                    ↓
   failed    parse_failed      chunk_failed       index_failed
                                                      ↓
                                                  retrying → indexed

  indexed → updating → re-indexing → indexed
  indexed → deleting → deleted

每个状态转换都发布事件（参考 249 事件驱动架构）。
```

## 最小示例

### 知识库 Agent 核心循环

```python
from dataclasses import dataclass, field
from enum import Enum
from datetime import datetime, timezone
from typing import Any
import uuid


class DocState(str, Enum):
    UPLOADED = "uploaded"
    PARSING = "parsing"
    PARSED = "parsed"
    CHUNKING = "chunking"
    CHUNKED = "chunked"
    INDEXING = "indexing"
    INDEXED = "indexed"
    FAILED = "failed"
    DELETING = "deleting"
    DELETED = "deleted"


@dataclass
class Document:
    doc_id: str
    title: str
    content: str
    state: DocState = DocState.UPLOADED
    chunks: list[str] = field(default_factory=list)
    error: str | None = None
    created_at: str = field(default_factory=lambda: datetime.now(timezone.utc).isoformat())
    indexed_at: str | None = None


@dataclass
class ToolResult:
    success: bool
    data: Any = None
    error: str | None = None


class KnowledgeBaseAgent:
    """知识库管理 Agent（最小版本）。"""

    def __init__(self):
        self.documents: dict[str, Document] = {}
        self.vector_store: dict[str, list[str]] = {}  # doc_id → chunks
        self.action_log: list[str] = []

    # === 工具实现 ===

    def upload_document(self, title: str, content: str) -> ToolResult:
        doc_id = f"DOC-{uuid.uuid4().hex[:8].upper()}"
        doc = Document(doc_id=doc_id, title=title, content=content)
        self.documents[doc_id] = doc
        self._log(f"上传文档: {title} ({doc_id})")
        return ToolResult(success=True, data={"doc_id": doc_id})

    def parse_and_chunk(self, doc_id: str) -> ToolResult:
        doc = self.documents.get(doc_id)
        if not doc:
            return ToolResult(success=False, error="文档不存在")

        doc.state = DocState.PARSING
        # 模拟解析
        if len(doc.content) < 10:
            doc.state = DocState.FAILED
            doc.error = "内容太短，无法解析"
            return ToolResult(success=False, error=doc.error)

        doc.state = DocState.CHUNKING
        # 简单分块：按段落
        doc.chunks = [p.strip() for p in doc.content.split("\n\n") if p.strip()]
        doc.state = DocState.CHUNKED
        self._log(f"分块完成: {doc_id} → {len(doc.chunks)} 块")
        return ToolResult(success=True, data={"chunks": len(doc.chunks)})

    def index_document(self, doc_id: str) -> ToolResult:
        doc = self.documents.get(doc_id)
        if not doc:
            return ToolResult(success=False, error="文档不存在")
        if doc.state != DocState.CHUNKED:
            return ToolResult(success=False, error=f"状态错误: {doc.state}")

        doc.state = DocState.INDEXING
        # 模拟写入向量库
        self.vector_store[doc_id] = doc.chunks
        doc.state = DocState.INDEXED
        doc.indexed_at = datetime.now(timezone.utc).isoformat()
        self._log(f"索引完成: {doc_id}")
        return ToolResult(success=True, data={"indexed_chunks": len(doc.chunks)})

    def delete_document(self, doc_id: str) -> ToolResult:
        doc = self.documents.get(doc_id)
        if not doc:
            return ToolResult(success=False, error="文档不存在")

        doc.state = DocState.DELETING
        # 清理向量
        self.vector_store.pop(doc_id, None)
        doc.state = DocState.DELETED
        self._log(f"删除文档: {doc_id}（含向量清理）")
        return ToolResult(success=True)

    def search(self, query: str, top_k: int = 3) -> ToolResult:
        results = []
        for doc_id, chunks in self.vector_store.items():
            for chunk in chunks:
                if query.lower() in chunk.lower():
                    results.append({"doc_id": doc_id, "chunk": chunk[:100]})
        return ToolResult(success=True, data=results[:top_k])

    def get_stats(self) -> ToolResult:
        total = len(self.documents)
        indexed = sum(1 for d in self.documents.values() if d.state == DocState.INDEXED)
        failed = sum(1 for d in self.documents.values() if d.state == DocState.FAILED)
        return ToolResult(success=True, data={
            "total_documents": total,
            "indexed": indexed,
            "failed": failed,
            "total_vectors": sum(len(v) for v in self.vector_store.values()),
        })

    # === Agent 循环 ===

    def process_new_document(self, title: str, content: str) -> dict:
        """完整的文档处理流水线。"""
        # Step 1: 上传
        r = self.upload_document(title, content)
        if not r.success:
            return {"status": "error", "step": "upload", "error": r.error}
        doc_id = r.data["doc_id"]

        # Step 2: 解析分块
        r = self.parse_and_chunk(doc_id)
        if not r.success:
            return {"status": "error", "step": "parse", "error": r.error, "doc_id": doc_id}

        # Step 3: 索引
        r = self.index_document(doc_id)
        if not r.success:
            return {"status": "error", "step": "index", "error": r.error, "doc_id": doc_id}

        return {"status": "success", "doc_id": doc_id, "chunks": r.data["indexed_chunks"]}

    def _log(self, msg: str):
        self.action_log.append(f"[{datetime.now(timezone.utc).strftime('%H:%M:%S')}] {msg}")


# 演示
def demo_kb_agent():
    agent = KnowledgeBaseAgent()

    # 处理文档
    result = agent.process_new_document(
        "RAG 系统架构",
        "RAG 系统由检索器和生成器组成。\n\n检索器负责从向量库中找到相关文档。\n\n生成器负责根据检索结果生成回答。"
    )
    print(f"  处理结果: {result}")

    # 搜索
    search_result = agent.search("检索器")
    print(f"  搜索结果: {search_result.data}")

    # 统计
    stats = agent.get_stats()
    print(f"  统计: {stats.data}")

    # 日志
    print(f"  操作日志: {len(agent.action_log)} 条")
```

## 工程化版本

### 带重试和降级的索引管道

```python
import asyncio
from dataclasses import dataclass
from typing import Callable


@dataclass
class RetryPolicy:
    max_retries: int = 3
    base_delay: float = 1.0
    max_delay: float = 30.0
    backoff_factor: float = 2.0

    def get_delay(self, attempt: int) -> float:
        delay = self.base_delay * (self.backoff_factor ** attempt)
        return min(delay, self.max_delay)


class IndexPipeline:
    """带重试的索引管道。"""

    def __init__(self, retry_policy: RetryPolicy | None = None):
        self.policy = retry_policy or RetryPolicy()
        self._steps: list[tuple[str, Callable]] = []

    def add_step(self, name: str, fn: Callable):
        self._steps.append((name, fn))

    async def execute(self, doc: Document) -> dict:
        """执行管道，每步支持重试。"""
        results = {}
        for step_name, step_fn in self._steps:
            success = False
            last_error = None

            for attempt in range(self.policy.max_retries + 1):
                try:
                    result = step_fn(doc)
                    results[step_name] = result
                    success = True
                    break
                except Exception as e:
                    last_error = str(e)
                    if attempt < self.policy.max_retries:
                        delay = self.policy.get_delay(attempt)
                        print(f"  [{step_name}] 重试 {attempt+1}/{self.policy.max_retries}，等待 {delay:.1f}s")
                        await asyncio.sleep(delay)

            if not success:
                doc.state = DocState.FAILED
                doc.error = f"{step_name}: {last_error}"
                return {"status": "failed", "step": step_name, "error": last_error}

        return {"status": "success", "results": results}


# 使用
async def demo_pipeline():
    pipeline = IndexPipeline(RetryPolicy(max_retries=2, base_delay=0.1))

    call_count = {"parse": 0}

    def flaky_parse(doc: Document) -> dict:
        call_count["parse"] += 1
        if call_count["parse"] < 3:
            raise ConnectionError("解析服务暂时不可用")
        doc.chunks = doc.content.split("\n\n")
        doc.state = DocState.CHUNKED
        return {"chunks": len(doc.chunks)}

    def index(doc: Document) -> dict:
        doc.state = DocState.INDEXED
        return {"vectors": len(doc.chunks)}

    pipeline.add_step("parse", flaky_parse)
    pipeline.add_step("index", index)

    doc = Document(doc_id="D-1", title="测试", content="段落1\n\n段落2\n\n段落3")
    result = await pipeline.execute(doc)
    print(f"  管道结果: {result}")
    print(f"  文档状态: {doc.state}")
```

### 增量更新与版本管理

```python
import hashlib


class DocumentVersionManager:
    """文档版本管理：检测变化，决定是否需要重新索引。"""

    def __init__(self):
        self._versions: dict[str, str] = {}  # doc_id → content_hash

    def compute_hash(self, content: str) -> str:
        return hashlib.sha256(content.encode()).hexdigest()[:16]

    def has_changed(self, doc_id: str, new_content: str) -> bool:
        """检测文档内容是否有变化。"""
        new_hash = self.compute_hash(new_content)
        old_hash = self._versions.get(doc_id)
        return old_hash != new_hash

    def record_version(self, doc_id: str, content: str):
        """记录当前版本。"""
        self._versions[doc_id] = self.compute_hash(content)

    def get_change_type(self, doc_id: str, new_content: str) -> str:
        """判断变化类型。"""
        if doc_id not in self._versions:
            return "new"
        if not self.has_changed(doc_id, new_content):
            return "unchanged"
        return "modified"


class IncrementalIndexer:
    """增量索引器：只处理有变化的文档。"""

    def __init__(self):
        self.version_mgr = DocumentVersionManager()
        self.indexed_docs: set[str] = set()

    def plan_sync(self, documents: list[dict]) -> dict:
        """规划同步操作。"""
        to_add = []
        to_update = []
        to_delete = []
        unchanged = []

        current_ids = set()
        for doc in documents:
            doc_id = doc["doc_id"]
            current_ids.add(doc_id)
            change = self.version_mgr.get_change_type(doc_id, doc["content"])

            if change == "new":
                to_add.append(doc_id)
            elif change == "modified":
                to_update.append(doc_id)
            else:
                unchanged.append(doc_id)

        # 检测删除（在索引中但不在源中）
        for indexed_id in self.indexed_docs:
            if indexed_id not in current_ids:
                to_delete.append(indexed_id)

        return {
            "to_add": to_add,
            "to_update": to_update,
            "to_delete": to_delete,
            "unchanged": len(unchanged),
        }

    def mark_synced(self, doc_id: str, content: str):
        self.version_mgr.record_version(doc_id, content)
        self.indexed_docs.add(doc_id)

    def mark_deleted(self, doc_id: str):
        self.indexed_docs.discard(doc_id)
```

### 孤立向量清理

```python
class OrphanCleaner:
    """清理向量库中的孤立数据。"""

    def __init__(self):
        self.document_registry: set[str] = set()  # 有效文档 ID
        self.vector_registry: set[str] = set()    # 向量库中的文档 ID

    def register_document(self, doc_id: str):
        self.document_registry.add(doc_id)

    def register_vectors(self, doc_id: str):
        self.vector_registry.add(doc_id)

    def remove_document(self, doc_id: str):
        self.document_registry.discard(doc_id)

    def find_orphans(self) -> list[str]:
        """找出有向量但没有文档记录的 ID。"""
        return list(self.vector_registry - self.document_registry)

    def find_missing(self) -> list[str]:
        """找出有文档但没有向量的 ID（索引缺失）。"""
        return list(self.document_registry - self.vector_registry)

    def cleanup(self) -> dict:
        """执行清理。"""
        orphans = self.find_orphans()
        for orphan_id in orphans:
            self.vector_registry.discard(orphan_id)

        missing = self.find_missing()
        return {
            "orphans_cleaned": len(orphans),
            "missing_vectors": missing,
            "orphan_ids": orphans,
        }
```

## 常见失败模式

### 1. 索引失败但文档标记为成功

```text
原因：
  索引写入超时，但代码没有正确处理异常

后果：
  用户搜索不到该文档的内容

解决：
  1. 索引后验证：写入后立即查询确认
  2. 状态机严格转换：只有验证通过才标记 indexed
  3. 定期一致性检查
```

### 2. 文档更新后旧向量残留

```text
原因：
  更新文档时只写了新向量，没删旧向量

后果：
  搜索返回过期内容

解决：
  更新流程：先删旧向量 → 再写新向量
  或者：用 doc_id + version 作为向量 ID，覆盖写入
```

### 3. 大文档 OOM

```text
原因：
  一次性加载 100MB PDF 到内存

后果：
  worker 崩溃，影响其他任务

解决：
  1. 流式解析：逐页处理
  2. 文件大小限制：拒绝 > 50MB
  3. 独立进程：大文档在子进程中处理
```

### 4. 并发索引冲突

```text
原因：
  两个 worker 同时索引同一个文档

后果：
  向量重复或数据不一致

解决：
  1. 分布式锁：索引前获取 doc_id 的锁
  2. 幂等设计：相同内容重复索引不产生副作用
  3. 乐观锁：用版本号检测冲突
```

## 什么时候不要这么做

```text
不需要知识库 Agent 的场景：

1. 文档数量极少（< 10 篇）
   手动管理完全够用

2. 文档几乎不更新
   一次性索引即可，不需要持续管理

3. 没有自动化需求
   如果团队有专人维护，Agent 反而增加复杂度

4. 实时性要求不高
   每周手动重建索引也能接受
```

### 健康检查设计

```python
class KBHealthChecker:
    """知识库健康检查。"""

    def __init__(self, agent: KnowledgeBaseAgent):
        self.agent = agent

    def run_full_check(self) -> dict:
        """执行全面健康检查。"""
        checks = {
            "document_integrity": self._check_documents(),
            "vector_consistency": self._check_vectors(),
            "stale_documents": self._check_stale(),
            "error_rate": self._check_errors(),
        }

        all_healthy = all(c["healthy"] for c in checks.values())
        return {
            "overall_healthy": all_healthy,
            "checks": checks,
            "timestamp": datetime.now(timezone.utc).isoformat(),
        }

    def _check_documents(self) -> dict:
        """检查文档完整性。"""
        total = len(self.agent.documents)
        failed = sum(1 for d in self.agent.documents.values() if d.state == DocState.FAILED)
        return {
            "healthy": failed == 0,
            "total": total,
            "failed": failed,
            "message": f"{failed}/{total} 文档索引失败" if failed else "所有文档正常",
        }

    def _check_vectors(self) -> dict:
        """检查向量一致性。"""
        indexed_docs = {
            doc_id for doc_id, d in self.agent.documents.items()
            if d.state == DocState.INDEXED
        }
        vector_docs = set(self.agent.vector_store.keys())

        orphans = vector_docs - indexed_docs
        missing = indexed_docs - vector_docs

        return {
            "healthy": len(orphans) == 0 and len(missing) == 0,
            "orphans": len(orphans),
            "missing": len(missing),
        }

    def _check_stale(self) -> dict:
        """检查过期文档（超过 30 天未更新）。"""
        # 简化：实际中比较 updated_at
        return {"healthy": True, "stale_count": 0}

    def _check_errors(self) -> dict:
        """检查错误率。"""
        total = len(self.agent.documents)
        if total == 0:
            return {"healthy": True, "error_rate": 0}
        failed = sum(1 for d in self.agent.documents.values() if d.state == DocState.FAILED)
        rate = failed / total
        return {
            "healthy": rate < 0.05,
            "error_rate": round(rate, 4),
        }
```

### 多租户隔离

```text
知识库 Agent 的多租户设计：

数据隔离：
  每个租户有独立的 namespace
  向量 ID 格式：{tenant_id}:{doc_id}:{chunk_idx}
  搜索时自动过滤 tenant_id

资源隔离：
  每个租户有索引配额（如最多 1000 篇文档）
  每个租户有 QPS 限制
  大租户独立 worker 池

权限隔离：
  Agent 工具调用时携带 tenant_id
  不允许跨租户访问
  审计日志记录所有操作
```

## 如何观测和评测

### 关键指标

```text
知识库 Agent 监控指标：

1. 索引延迟
   文档上传到可搜索的时间
   目标：< 5 分钟

2. 索引成功率
   成功索引 / 总上传数
   目标：> 99%

3. 搜索命中率
   搜索返回结果 / 总搜索次数
   目标：> 90%

4. 向量一致性
   有文档无向量 + 有向量无文档 = 0
   目标：每天检查，不一致 = 0

5. 孤立向量比例
   孤立向量 / 总向量
   目标：< 1%
```

### 评测方法

```python
class KBEvaluator:
    """知识库质量评测。"""

    def __init__(self, agent: KnowledgeBaseAgent):
        self.agent = agent

    def evaluate_retrieval(self, test_cases: list[dict]) -> dict:
        """评测检索质量。"""
        hits = 0
        total = len(test_cases)

        for case in test_cases:
            result = self.agent.search(case["query"], top_k=5)
            found = any(
                case["expected_doc_id"] in str(r)
                for r in (result.data or [])
            )
            if found:
                hits += 1

        return {
            "recall_at_5": round(hits / max(total, 1), 4),
            "total_cases": total,
            "hits": hits,
        }

    def evaluate_freshness(self) -> dict:
        """评测内容新鲜度。"""
        indexed = [
            d for d in self.agent.documents.values()
            if d.state == DocState.INDEXED
        ]
        if not indexed:
            return {"freshness": "N/A"}

        # 简化：实际中比较 indexed_at 和 source_updated_at
        return {
            "indexed_count": len(indexed),
            "all_have_timestamp": all(d.indexed_at for d in indexed),
        }
```

### 定时任务调度

```python
class KBScheduler:
    """知识库定时任务调度器。"""

    def __init__(self, agent: KnowledgeBaseAgent, cleaner: OrphanCleaner):
        self.agent = agent
        self.cleaner = cleaner
        self._tasks: list[dict] = []

    def register_task(self, name: str, interval_seconds: int, fn):
        """注册定时任务。"""
        self._tasks.append({
            "name": name,
            "interval": interval_seconds,
            "fn": fn,
            "last_run": None,
            "run_count": 0,
        })

    async def run_loop(self, max_iterations: int = 10):
        """调度主循环。"""
        import time
        for _ in range(max_iterations):
            now = time.time()
            for task in self._tasks:
                if task["last_run"] is None or (now - task["last_run"]) >= task["interval"]:
                    print(f"  [调度] 执行: {task['name']}")
                    try:
                        task["fn"]()
                        task["run_count"] += 1
                    except Exception as e:
                        print(f"  [调度] 失败: {task['name']} → {e}")
                    task["last_run"] = now
            await asyncio.sleep(1)

    def get_schedule_status(self) -> list[dict]:
        return [
            {"name": t["name"], "interval": t["interval"], "runs": t["run_count"]}
            for t in self._tasks
        ]


# 使用示例
def setup_scheduler():
    agent = KnowledgeBaseAgent()
    cleaner = OrphanCleaner()
    scheduler = KBScheduler(agent, cleaner)

    # 每 5 分钟：健康检查
    scheduler.register_task("health_check", 300, lambda: None)
    # 每 30 分钟：清理孤立向量
    scheduler.register_task("orphan_cleanup", 1800, cleaner.cleanup)
    # 每 1 小时：统计报告
    scheduler.register_task("stats_report", 3600, agent.get_stats)

    return scheduler
```

```python
from abc import ABC, abstractmethod


class DocumentParser(ABC):
    """文档解析器基类。"""

    @abstractmethod
    def can_parse(self, file_type: str) -> bool:
        ...

    @abstractmethod
    def parse(self, content: bytes) -> list[str]:
        """返回段落列表。"""
        ...


class MarkdownParser(DocumentParser):
    def can_parse(self, file_type: str) -> bool:
        return file_type in ("md", "markdown")

    def parse(self, content: bytes) -> list[str]:
        text = content.decode("utf-8")
        # 按标题分块
        sections = []
        current = []
        for line in text.split("\n"):
            if line.startswith("#") and current:
                sections.append("\n".join(current))
                current = []
            current.append(line)
        if current:
            sections.append("\n".join(current))
        return [s for s in sections if s.strip()]


class PdfParser(DocumentParser):
    def can_parse(self, file_type: str) -> bool:
        return file_type == "pdf"

    def parse(self, content: bytes) -> list[str]:
        # 实际中用 PyPDF2 / pdfplumber
        text = content.decode("utf-8", errors="ignore")
        pages = text.split("\f")  # 分页符
        return [p.strip() for p in pages if p.strip()]


class PlainTextParser(DocumentParser):
    def can_parse(self, file_type: str) -> bool:
        return file_type in ("txt", "text")

    def parse(self, content: bytes) -> list[str]:
        text = content.decode("utf-8")
        paragraphs = text.split("\n\n")
        return [p.strip() for p in paragraphs if p.strip()]
```
