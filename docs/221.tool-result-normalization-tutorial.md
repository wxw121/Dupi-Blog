# AI Agent 工程（八）：工具结果标准化

> 工具返回的原始数据对 LLM 来说往往是噪音：字段太多、格式不一、关键信息淹没在嵌套结构中。标准化是把"机器能读"变成"模型能用"。

**环境要求：**
- Python **3.10+**
- 无需安装第三方库

## 你会学到什么

- 设计统一的工具返回信封格式。
- 将不同数据源的返回值归一化。
- 控制返回给 LLM 的信息量（token 预算）。
- 错误信息的标准化表达。
- 让 LLM 根据结果决定下一步行动。

## 它解决什么问题

```text
没有标准化时的问题：

1. 格式不一致
   → 搜索工具返回 {"results": [...]}
   → 数据库工具返回 [{"id": 1, ...}]
   → API 工具返回 {"data": {"items": [...]}}
   LLM 每次都要"猜"怎么解析

2. 信息过载
   → 数据库返回 50 个字段
   → LLM 只需要 3 个
   → 浪费 token，干扰推理

3. 错误格式不统一
   → 有的返回 {"error": "..."}
   → 有的返回 {"code": 500, "msg": "..."}
   → 有的直接抛异常
   LLM 不知道调用是否成功

4. 缺少元信息
   → 不知道结果是否完整
   → 不知道数据的新鲜度
   → 不知道下一步可以做什么
```

## 标准化信封格式

```text
统一的工具返回格式（ToolResult Envelope）：

{
  "success": true/false,         ← 调用是否成功
  "data": {...},                 ← 业务数据（成功时）
  "error": {                     ← 错误信息（失败时）
    "code": "NOT_FOUND",
    "message": "Order ORD-123 not found",
    "suggestion": "Check the order ID format"
  },
  "metadata": {                  ← 元信息
    "tool": "get_order",
    "latency_ms": 150,
    "truncated": false,
    "total_count": 1,
    "next_actions": ["cancel_order", "get_shipping"]
  }
}

设计原则：
1. success 字段必须存在 → LLM 一眼判断成败
2. data 结构按工具类型固定 → LLM 可以学习模式
3. error 包含修正建议 → 支持自修正
4. metadata 提供上下文 → 辅助决策
```

### 标准化后的返回

```python
from dataclasses import dataclass, field
from typing import Any, Optional
from enum import Enum
import time


class ErrorCode(str, Enum):
    """标准错误码"""
    NOT_FOUND = "NOT_FOUND"
    INVALID_INPUT = "INVALID_INPUT"
    PERMISSION_DENIED = "PERMISSION_DENIED"
    TIMEOUT = "TIMEOUT"
    RATE_LIMITED = "RATE_LIMITED"
    INTERNAL_ERROR = "INTERNAL_ERROR"
    DEPENDENCY_ERROR = "DEPENDENCY_ERROR"


@dataclass
class ToolError:
    """标准化错误"""
    code: ErrorCode
    message: str
    suggestion: str = ""
    retryable: bool = False


@dataclass
class ToolMetadata:
    """工具调用元信息"""
    tool_name: str
    latency_ms: float = 0.0
    truncated: bool = False
    total_count: int = 0
    returned_count: int = 0
    next_actions: list[str] = field(default_factory=list)
    cache_hit: bool = False


@dataclass
class StandardToolResult:
    """标准化工具返回"""
    success: bool
    data: Any = None
    error: Optional[ToolError] = None
    metadata: Optional[ToolMetadata] = None

    def to_llm_string(self) -> str:
        """转换为 LLM 友好的文本"""
        if not self.success:
            lines = [f"ERROR [{self.error.code.value}]: {self.error.message}"]
            if self.error.suggestion:
                lines.append(f"Suggestion: {self.error.suggestion}")
            if self.error.retryable:
                lines.append("This error is retryable.")
            return "\n".join(lines)

        lines = []
        if isinstance(self.data, dict):
            for k, v in self.data.items():
                lines.append(f"{k}: {v}")
        elif isinstance(self.data, list):
            for i, item in enumerate(self.data):
                lines.append(f"[{i+1}] {item}")
        else:
            lines.append(str(self.data))

        if self.metadata:
            if self.metadata.truncated:
                lines.append(
                    f"(Showing {self.metadata.returned_count} of "
                    f"{self.metadata.total_count} total results)"
                )
            if self.metadata.next_actions:
                actions = ", ".join(self.metadata.next_actions)
                lines.append(f"Available next actions: {actions}")

        return "\n".join(lines)
```

### 标准化工具包装器

```python
class ToolNormalizer:
    """将任意工具返回值标准化"""

    @staticmethod
    def wrap_success(
        data: Any,
        tool_name: str,
        latency_ms: float = 0,
        next_actions: list[str] | None = None,
    ) -> StandardToolResult:
        return StandardToolResult(
            success=True,
            data=data,
            metadata=ToolMetadata(
                tool_name=tool_name,
                latency_ms=latency_ms,
                next_actions=next_actions or [],
            ),
        )

    @staticmethod
    def wrap_error(
        code: ErrorCode,
        message: str,
        tool_name: str,
        suggestion: str = "",
        retryable: bool = False,
    ) -> StandardToolResult:
        return StandardToolResult(
            success=False,
            error=ToolError(
                code=code,
                message=message,
                suggestion=suggestion,
                retryable=retryable,
            ),
            metadata=ToolMetadata(tool_name=tool_name),
        )

    @staticmethod
    def wrap_exception(
        exc: Exception,
        tool_name: str,
    ) -> StandardToolResult:
        """将异常转换为标准错误"""
        if isinstance(exc, TimeoutError):
            return ToolNormalizer.wrap_error(
                ErrorCode.TIMEOUT,
                f"Tool '{tool_name}' timed out",
                tool_name,
                suggestion="Try again or use a simpler query",
                retryable=True,
            )
        if isinstance(exc, PermissionError):
            return ToolNormalizer.wrap_error(
                ErrorCode.PERMISSION_DENIED,
                str(exc),
                tool_name,
                suggestion="Check user permissions",
            )
        return ToolNormalizer.wrap_error(
            ErrorCode.INTERNAL_ERROR,
            f"Unexpected error: {type(exc).__name__}: {exc}",
            tool_name,
            retryable=False,
        )
```

## 工程化版本

### 数据裁剪器

```python
class DataTrimmer:
    """控制返回给 LLM 的数据量"""

    def __init__(self, max_tokens: int = 2000):
        self.max_tokens = max_tokens
        # 粗略估算：1 token ≈ 4 字符（英文）或 2 字符（中文）
        self.max_chars = max_tokens * 3

    def trim_dict(self, data: dict, priority_fields: list[str]) -> dict:
        """按优先级保留字段"""
        result = {}
        current_chars = 0

        # 先放高优先级字段
        for field_name in priority_fields:
            if field_name in data:
                field_str = str(data[field_name])
                if current_chars + len(field_str) <= self.max_chars:
                    result[field_name] = data[field_name]
                    current_chars += len(field_str)

        # 再放其他字段（如果还有空间）
        for key, value in data.items():
            if key not in result:
                value_str = str(value)
                if current_chars + len(value_str) <= self.max_chars:
                    result[key] = value
                    current_chars += len(value_str)

        return result

    def trim_list(
        self, items: list, max_items: int = 10
    ) -> tuple[list, bool]:
        """截断列表"""
        if len(items) <= max_items:
            return items, False
        return items[:max_items], True

    def trim_nested(self, data: Any, max_depth: int = 3) -> Any:
        """限制嵌套深度"""
        if max_depth <= 0:
            if isinstance(data, (dict, list)):
                return f"<{type(data).__name__} with {len(data)} items>"
            return data

        if isinstance(data, dict):
            return {
                k: self.trim_nested(v, max_depth - 1)
                for k, v in data.items()
            }
        if isinstance(data, list):
            return [self.trim_nested(item, max_depth - 1) for item in data]
        return data
```

### 格式适配器

```python
class FormatAdapter:
    """将不同数据源格式适配为标准格式"""

    @staticmethod
    def from_sql_rows(rows: list[dict], tool_name: str) -> StandardToolResult:
        """SQL 查询结果 → 标准格式"""
        if not rows:
            return ToolNormalizer.wrap_error(
                ErrorCode.NOT_FOUND,
                "No records found",
                tool_name,
                suggestion="Try broadening your search criteria",
            )
        return ToolNormalizer.wrap_success(
            data={"records": rows, "count": len(rows)},
            tool_name=tool_name,
        )

    @staticmethod
    def from_http_response(
        status_code: int, body: dict, tool_name: str
    ) -> StandardToolResult:
        """HTTP 响应 → 标准格式"""
        if status_code == 200:
            return ToolNormalizer.wrap_success(data=body, tool_name=tool_name)
        elif status_code == 404:
            return ToolNormalizer.wrap_error(
                ErrorCode.NOT_FOUND,
                body.get("message", "Resource not found"),
                tool_name,
            )
        elif status_code == 429:
            return ToolNormalizer.wrap_error(
                ErrorCode.RATE_LIMITED,
                "API rate limit exceeded",
                tool_name,
                suggestion="Wait a moment and try again",
                retryable=True,
            )
        else:
            return ToolNormalizer.wrap_error(
                ErrorCode.DEPENDENCY_ERROR,
                f"External API returned {status_code}",
                tool_name,
                retryable=status_code >= 500,
            )

    @staticmethod
    def from_exception(exc: Exception, tool_name: str) -> StandardToolResult:
        """异常 → 标准格式"""
        return ToolNormalizer.wrap_exception(exc, tool_name)
```

### 完整标准化执行器

```python
import asyncio
from typing import Callable


class NormalizedToolExecutor:
    """带标准化的工具执行器"""

    def __init__(self, max_tokens: int = 2000):
        self._tools: dict[str, Callable] = {}
        self._priority_fields: dict[str, list[str]] = {}
        self._next_actions: dict[str, list[str]] = {}
        self.trimmer = DataTrimmer(max_tokens)

    def register(
        self,
        name: str,
        handler: Callable,
        priority_fields: list[str] | None = None,
        next_actions: list[str] | None = None,
    ):
        self._tools[name] = handler
        self._priority_fields[name] = priority_fields or []
        self._next_actions[name] = next_actions or []

    async def execute(self, tool_name: str, params: dict) -> StandardToolResult:
        """执行工具并标准化返回"""
        if tool_name not in self._tools:
            return ToolNormalizer.wrap_error(
                ErrorCode.NOT_FOUND,
                f"Tool '{tool_name}' not registered",
                tool_name,
                suggestion=f"Available tools: {list(self._tools.keys())}",
            )

        start = time.perf_counter()
        try:
            handler = self._tools[tool_name]
            if asyncio.iscoroutinefunction(handler):
                raw_result = await handler(**params)
            else:
                raw_result = handler(**params)

            latency_ms = (time.perf_counter() - start) * 1000

            # 裁剪数据
            if isinstance(raw_result, dict):
                raw_result = self.trimmer.trim_dict(
                    raw_result,
                    self._priority_fields.get(tool_name, []),
                )
            elif isinstance(raw_result, list):
                raw_result, truncated = self.trimmer.trim_list(raw_result)
            else:
                truncated = False

            return StandardToolResult(
                success=True,
                data=raw_result,
                metadata=ToolMetadata(
                    tool_name=tool_name,
                    latency_ms=round(latency_ms, 2),
                    truncated=truncated if isinstance(raw_result, list) else False,
                    total_count=len(raw_result) if isinstance(raw_result, list) else 1,
                    returned_count=len(raw_result) if isinstance(raw_result, list) else 1,
                    next_actions=self._next_actions.get(tool_name, []),
                ),
            )

        except Exception as exc:
            return ToolNormalizer.wrap_exception(exc, tool_name)
```

## 常见失败模式

### 1. 返回原始 JSON

```text
症状：
  LLM 收到 5000 token 的 JSON，推理质量下降

原因：
  直接把 API 响应透传给 LLM

解决：
  1. 只保留 LLM 需要的字段
  2. 扁平化嵌套结构
  3. 用自然语言 key 替代缩写
  4. 控制总 token 数 < 2000
```

### 2. 成功和失败格式不同

```text
症状：
  LLM 不确定调用是否成功

原因：
  成功返回数据，失败返回 None
  或者成功返回 200，失败也返回 200 + error 字段

解决：
  success 字段必须在顶层
  失败时 data 为 null，error 有值
  成功时 error 为 null，data 有值
```

### 3. 列表结果没有截断

```text
症状：
  搜索返回 100 条结果，超出上下文窗口

原因：
  没有 max_items 限制

解决：
  默认最多返回 10 条
  metadata 中标注 truncated 和 total_count
  提供 next_actions: ["search_page_2"]
```

### 4. 错误没有可操作性

```text
症状：
  LLM 收到错误后不知道怎么办

原因：
  错误消息只有 "Failed"
  没有建议下一步

解决：
  error.suggestion 告诉 LLM 怎么修
  error.retryable 告诉 LLM 能否重试
  metadata.next_actions 提供替代方案
```

## 什么时候不要这么做

```text
不需要完整标准化的场景：

1. 工具只返回简单值
   → "42" 或 "true" 不需要信封包装
   → 直接返回即可

2. 内部管道（非 LLM 消费）
   → 代码间调用不需要 LLM 友好格式
   → 用原生数据结构更高效

3. 流式输出
   → 流式场景用增量格式
   → 不需要完整信封

4. 调试模式
   → 开发者需要看完整原始数据
   → 标准化会丢失信息
```

## 生产环境注意事项

### Token 预算管理

```python
class TokenBudgetManager:
    """管理工具返回的 token 预算"""

    def __init__(self, total_budget: int = 4000):
        self.total_budget = total_budget
        self._used: dict[str, int] = {}

    def allocate(self, tool_name: str, estimated_tokens: int) -> int:
        """为工具分配 token 预算"""
        remaining = self.total_budget - sum(self._used.values())
        allocated = min(estimated_tokens, remaining)
        self._used[tool_name] = allocated
        return allocated

    def get_remaining(self) -> int:
        return self.total_budget - sum(self._used.values())

    def report(self) -> dict:
        return {
            "total": self.total_budget,
            "used": sum(self._used.values()),
            "remaining": self.get_remaining(),
            "per_tool": self._used,
        }
```

### 结果缓存

```python
from hashlib import sha256
import json


class ToolResultCache:
    """工具结果缓存（避免重复调用）"""

    def __init__(self, ttl_seconds: int = 300):
        self._cache: dict[str, tuple[float, StandardToolResult]] = {}
        self.ttl = ttl_seconds

    def _make_key(self, tool_name: str, params: dict) -> str:
        raw = f"{tool_name}:{json.dumps(params, sort_keys=True)}"
        return sha256(raw.encode()).hexdigest()

    def get(self, tool_name: str, params: dict) -> StandardToolResult | None:
        key = self._make_key(tool_name, params)
        if key not in self._cache:
            return None
        timestamp, result = self._cache[key]
        if time.time() - timestamp > self.ttl:
            del self._cache[key]
            return None
        # 标记缓存命中
        if result.metadata:
            result.metadata.cache_hit = True
        return result

    def put(self, tool_name: str, params: dict, result: StandardToolResult):
        # 只缓存成功的结果
        if result.success:
            key = self._make_key(tool_name, params)
            self._cache[key] = (time.time(), result)
```

### 敏感信息脱敏

```python
class SensitiveDataRedactor:
    """工具返回中的敏感信息脱敏"""

    SENSITIVE_PATTERNS = {
        "phone": (r"1[3-9]\d{9}", "1**********"),
        "email": (r"[\w.]+@[\w.]+", "***@***.com"),
        "id_card": (r"\d{17}[\dXx]", "******************"),
        "bank_card": (r"\d{16,19}", "**** **** **** ****"),
    }

    @classmethod
    def redact(cls, data: Any) -> Any:
        """递归脱敏"""
        import re
        if isinstance(data, str):
            for name, (pattern, replacement) in cls.SENSITIVE_PATTERNS.items():
                data = re.sub(pattern, replacement, data)
            return data
        if isinstance(data, dict):
            return {k: cls.redact(v) for k, v in data.items()}
        if isinstance(data, list):
            return [cls.redact(item) for item in data]
        return data
```

## 如何观测和评测

### 标准化质量指标

```text
评估标准化效果的指标：

1. Token 效率
   - 标准化前平均 token 数
   - 标准化后平均 token 数
   - 压缩率 = 1 - (后/前)
   - 目标：压缩率 > 50%

2. LLM 理解准确率
   - 给 LLM 标准化结果，问"调用成功了吗？"
   - 给 LLM 标准化结果，问"下一步可以做什么？"
   - 目标：准确率 > 95%

3. 决策质量
   - LLM 根据结果做出的下一步决策是否合理
   - 对比标准化前后的决策质量
   - 目标：合理率提升 > 20%

4. 错误恢复率
   - 收到错误后 LLM 能否自修正
   - 有 suggestion vs 无 suggestion 的对比
   - 目标：自修正成功率 > 70%
```

## 和 RAG / 后端 / 前端的关系

```text
结果标准化在系统中的位置：

RAG 管道：
  检索结果需要标准化为统一格式
  包含：content, source, score, metadata
  控制返回 chunk 数量（token 预算）
  标注是否有更多结果可查

后端 API：
  后端返回完整数据（面向前端渲染）
  Tool 层做裁剪（面向 LLM 推理）
  两层独立，不互相影响

前端展示：
  前端用后端 API 的完整数据渲染
  Agent 的推理过程可以展示给用户
  标准化结果中的 next_actions 可以渲染为按钮
```

## 面试怎么讲

```text
面试官：工具返回值你怎么处理？

回答框架（2 分钟）：

"我设计了一个标准化信封格式：

 结构：success + data + error + metadata
 - success：LLM 一眼判断成败
 - data：裁剪后的业务数据
 - error：错误码 + 消息 + 修正建议
 - metadata：延迟、截断信息、下一步操作

 关键设计：
 1. Token 预算控制：每个工具返回不超过 2000 token
 2. 字段优先级：只保留 LLM 决策需要的字段
 3. 错误可操作：每个错误带 suggestion 和 retryable
 4. next_actions：告诉 LLM 接下来可以做什么

 实际效果：
 工具返回 token 减少 60%，
 LLM 决策准确率提升 25%，
 错误自修正成功率 72%。"

追问 1：怎么决定保留哪些字段？
  → 按工具类型定义 priority_fields。
    例：订单工具保留 status, items, amount；
    去掉 internal_notes, warehouse_code。

追问 2：结果太大怎么办？
  → 三级裁剪：字段过滤 → 列表截断 → 深度限制。
    metadata 中标注 truncated，LLM 可以请求更多。

追问 3：缓存怎么做？
  → 对读操作做 TTL 缓存（5 分钟）。
    写操作不缓存。
    缓存命中标注在 metadata 中。
```

## 实战案例：多工具结果聚合

### 场景：Agent 并行调用多个工具

```python
import asyncio
from dataclasses import dataclass, field


@dataclass
class AggregatedResult:
    """多工具聚合结果"""
    results: dict[str, StandardToolResult] = field(default_factory=dict)
    total_latency_ms: float = 0.0
    all_success: bool = True

    def summary_for_llm(self) -> str:
        """生成聚合摘要"""
        lines = []
        lines.append(f"Executed {len(self.results)} tools "
                     f"in {self.total_latency_ms:.0f}ms:")
        lines.append("")

        for tool_name, result in self.results.items():
            if result.success:
                lines.append(f"✓ {tool_name}: success")
                # 只展示摘要
                data_str = str(result.data)
                if len(data_str) > 200:
                    data_str = data_str[:200] + "..."
                lines.append(f"  {data_str}")
            else:
                lines.append(f"✗ {tool_name}: {result.error.code.value}")
                lines.append(f"  {result.error.message}")
            lines.append("")

        if not self.all_success:
            failed = [n for n, r in self.results.items() if not r.success]
            lines.append(f"Failed tools: {', '.join(failed)}")
            lines.append("Consider alternative approaches for failed tools.")

        return "\n".join(lines)


class ParallelToolExecutor:
    """并行工具执行器"""

    def __init__(self, executor: NormalizedToolExecutor):
        self.executor = executor

    async def execute_parallel(
        self, calls: list[tuple[str, dict]]
    ) -> AggregatedResult:
        """并行执行多个工具调用"""
        start = time.perf_counter()

        # 并行执行
        tasks = [
            self.executor.execute(tool_name, params)
            for tool_name, params in calls
        ]
        results = await asyncio.gather(*tasks, return_exceptions=True)

        # 组装结果
        aggregated = AggregatedResult()
        for (tool_name, _), result in zip(calls, results):
            if isinstance(result, Exception):
                aggregated.results[tool_name] = ToolNormalizer.wrap_exception(
                    result, tool_name
                )
            else:
                aggregated.results[tool_name] = result

        aggregated.total_latency_ms = (time.perf_counter() - start) * 1000
        aggregated.all_success = all(
            r.success for r in aggregated.results.values()
        )
        return aggregated


# 使用示例
async def research_query(query: str):
    """研究查询：并行调用多个工具"""
    executor = NormalizedToolExecutor(max_tokens=1500)
    # 注册工具...

    parallel = ParallelToolExecutor(executor)
    result = await parallel.execute_parallel([
        ("search_knowledge_base", {"query": query, "top_k": 3}),
        ("search_web", {"query": query, "max_results": 5}),
        ("get_related_tickets", {"query": query, "limit": 3}),
    ])

    # 给 LLM 聚合摘要
    return result.summary_for_llm()
```

### 结果转换链

```python
class ResultTransformChain:
    """结果转换链：按顺序应用多个转换器"""

    def __init__(self):
        self._transforms: list[Callable] = []

    def add(self, transform: Callable[[StandardToolResult], StandardToolResult]):
        """添加转换器"""
        self._transforms.append(transform)
        return self

    def apply(self, result: StandardToolResult) -> StandardToolResult:
        """依次应用所有转换"""
        for transform in self._transforms:
            result = transform(result)
        return result


# 常用转换器
def redact_sensitive(result: StandardToolResult) -> StandardToolResult:
    """脱敏转换"""
    if result.data:
        result.data = SensitiveDataRedactor.redact(result.data)
    return result


def flatten_nested(result: StandardToolResult) -> StandardToolResult:
    """扁平化嵌套结构"""
    if isinstance(result.data, dict):
        result.data = _flatten_dict(result.data)
    return result


def _flatten_dict(d: dict, prefix: str = "") -> dict:
    """递归扁平化字典"""
    items = {}
    for k, v in d.items():
        new_key = f"{prefix}.{k}" if prefix else k
        if isinstance(v, dict):
            items.update(_flatten_dict(v, new_key))
        else:
            items[new_key] = v
    return items


def add_context_hints(result: StandardToolResult) -> StandardToolResult:
    """添加上下文提示"""
    if result.success and result.metadata:
        if result.metadata.latency_ms > 5000:
            if not result.metadata.next_actions:
                result.metadata.next_actions = []
            result.metadata.next_actions.append(
                "Note: This tool was slow. Consider caching."
            )
    return result


# 组装转换链
def build_default_chain() -> ResultTransformChain:
    """构建默认转换链"""
    chain = ResultTransformChain()
    chain.add(redact_sensitive)
    chain.add(flatten_nested)
    chain.add(add_context_hints)
    return chain
```

## 设计原则总结

```text
工具结果标准化 7 条原则：

1. 统一信封格式
   所有工具返回 success + data + error + metadata
   LLM 不需要学习每种工具的独特格式

2. 控制 token 预算
   每个工具返回不超过 2000 token
   超出就裁剪，标注 truncated

3. 错误可操作
   每个错误带 code + message + suggestion
   LLM 能根据错误决定下一步

4. 提供 next_actions
   告诉 LLM 接下来可以做什么
   减少 LLM 的探索空间

5. 脱敏先行
   返回给 LLM 前必须脱敏
   手机号、身份证、银行卡不能明文

6. 缓存读操作
   相同参数的读操作缓存 5 分钟
   写操作不缓存

7. 日志全记录
   每次调用记录：工具名、参数、结果、延迟
   用于事后分析和问题排查
```
