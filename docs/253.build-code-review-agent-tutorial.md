## 它解决什么问题

### 人工 Code Review 的痛点

```text
问题：
  1. 审查者时间不够 → 走马观花
  2. 风格问题占用大量精力 → 忽略逻辑 bug
  3. 跨文件影响难以发现
  4. 新人 PR 等待时间长
  5. 审查标准不一致

结果：
  - 明显 bug 漏过
  - 审查变成"LGTM 盖章"
  - 团队效率下降
```

### Agent 审查的价值

```text
Agent 擅长：
  ✓ 一致性检查（命名、格式、模式）
  ✓ 常见 bug 模式（空指针、资源泄漏）
  ✓ 安全漏洞（SQL 注入、XSS）
  ✓ 性能问题（N+1 查询、内存泄漏）
  ✓ 跨文件依赖分析

Agent 不擅长：
  ✗ 业务逻辑正确性（需要领域知识）
  ✗ 架构设计决策
  ✗ 代码"优雅度"的主观判断

定位：Agent 做第一轮筛选，人做最终决策。
```

## 核心概念：代码审查 Agent 的审查维度

### 审查分类

```text
Severity 1 - Critical（必须修复）：
  - 安全漏洞（注入、越权）
  - 数据丢失风险
  - 死锁 / 无限循环
  - 生产事故级 bug

Severity 2 - Major（强烈建议修复）：
  - 逻辑错误
  - 资源泄漏
  - 并发问题
  - 错误处理缺失

Severity 3 - Minor（建议改进）：
  - 命名不规范
  - 重复代码
  - 缺少注释
  - 可简化的逻辑

Severity 4 - Suggestion（可选优化）：
  - 更好的设计模式
  - 性能微优化
  - 代码风格偏好
```

### 审查流程

```text
1. 接收 PR / Diff
2. 解析变更文件列表
3. 对每个文件：
   a. 静态分析（规则引擎）
   b. 上下文理解（读取相关文件）
   c. LLM 审查（逻辑、设计）
4. 汇总发现
5. 去重 + 排序（按严重度）
6. 生成审查报告
7. 发布评论（inline comments）
```

## 最小示例

### 代码审查 Agent 核心

```python
from dataclasses import dataclass, field
from enum import Enum
from typing import Any
import re


class Severity(str, Enum):
    CRITICAL = "critical"
    MAJOR = "major"
    MINOR = "minor"
    SUGGESTION = "suggestion"


class Category(str, Enum):
    SECURITY = "security"
    BUG = "bug"
    PERFORMANCE = "performance"
    STYLE = "style"
    DESIGN = "design"
    CONCURRENCY = "concurrency"


@dataclass
class ReviewComment:
    """审查意见。"""
    file_path: str
    line_number: int
    severity: Severity
    category: Category
    message: str
    suggestion: str = ""
    code_snippet: str = ""


@dataclass
class FileDiff:
    """文件变更。"""
    file_path: str
    added_lines: list[tuple[int, str]]  # (行号, 内容)
    removed_lines: list[tuple[int, str]]
    language: str = "python"


@dataclass
class ReviewReport:
    """审查报告。"""
    pr_title: str
    files_reviewed: int
    comments: list[ReviewComment]
    summary: str
    approved: bool

    @property
    def critical_count(self) -> int:
        return sum(1 for c in self.comments if c.severity == Severity.CRITICAL)

    @property
    def major_count(self) -> int:
        return sum(1 for c in self.comments if c.severity == Severity.MAJOR)


class CodeReviewAgent:
    """代码审查 Agent（最小版本）。"""

    def __init__(self):
        self.rules: list[dict] = self._default_rules()

    def review_diff(self, pr_title: str, diffs: list[FileDiff]) -> ReviewReport:
        """审查一组文件变更。"""
        all_comments: list[ReviewComment] = []

        for diff in diffs:
            comments = self._review_file(diff)
            all_comments.extend(comments)

        # 排序：严重度优先
        severity_order = {Severity.CRITICAL: 0, Severity.MAJOR: 1, Severity.MINOR: 2, Severity.SUGGESTION: 3}
        all_comments.sort(key=lambda c: severity_order[c.severity])

        # 生成摘要
        summary = self._generate_summary(all_comments)

        # 是否通过
        approved = all(c.severity not in (Severity.CRITICAL, Severity.MAJOR) for c in all_comments)

        return ReviewReport(
            pr_title=pr_title,
            files_reviewed=len(diffs),
            comments=all_comments,
            summary=summary,
            approved=approved,
        )

    def _review_file(self, diff: FileDiff) -> list[ReviewComment]:
        """审查单个文件。"""
        comments = []

        for line_no, line in diff.added_lines:
            # 规则检查
            for rule in self.rules:
                if rule["pattern"] in line:
                    comments.append(ReviewComment(
                        file_path=diff.file_path,
                        line_number=line_no,
                        severity=rule["severity"],
                        category=rule["category"],
                        message=rule["message"],
                        suggestion=rule.get("suggestion", ""),
                        code_snippet=line.strip(),
                    ))

        return comments

    def _generate_summary(self, comments: list[ReviewComment]) -> str:
        """生成审查摘要。"""
        if not comments:
            return "代码质量良好，未发现问题。✓"

        parts = []
        critical = sum(1 for c in comments if c.severity == Severity.CRITICAL)
        major = sum(1 for c in comments if c.severity == Severity.MAJOR)
        minor = sum(1 for c in comments if c.severity == Severity.MINOR)

        if critical:
            parts.append(f"🔴 {critical} 个严重问题")
        if major:
            parts.append(f"🟠 {major} 个重要问题")
        if minor:
            parts.append(f"🟡 {minor} 个小问题")

        return f"发现 {'，'.join(parts)}。" if parts else "审查通过。"

    def _default_rules(self) -> list[dict]:
        """默认规则集。"""
        return [
            {
                "pattern": "eval(",
                "severity": Severity.CRITICAL,
                "category": Category.SECURITY,
                "message": "使用 eval() 存在代码注入风险",
                "suggestion": "使用 ast.literal_eval() 或 JSON 解析替代",
            },
            {
                "pattern": "exec(",
                "severity": Severity.CRITICAL,
                "category": Category.SECURITY,
                "message": "使用 exec() 存在代码注入风险",
                "suggestion": "避免动态执行代码，使用白名单映射",
            },
            {
                "pattern": "password",
                "severity": Severity.CRITICAL,
                "category": Category.SECURITY,
                "message": "疑似硬编码密码",
                "suggestion": "使用环境变量或密钥管理服务",
            },
            {
                "pattern": "except:",
                "severity": Severity.MAJOR,
                "category": Category.BUG,
                "message": "裸 except 会吞掉所有异常（包括 KeyboardInterrupt）",
                "suggestion": "使用 except Exception as e: 或更具体的异常类型",
            },
            {
                "pattern": "TODO",
                "severity": Severity.MINOR,
                "category": Category.STYLE,
                "message": "存在未完成的 TODO",
                "suggestion": "在合并前处理或创建 issue 跟踪",
            },
            {
                "pattern": "print(",
                "severity": Severity.MINOR,
                "category": Category.STYLE,
                "message": "生产代码中不应使用 print()",
                "suggestion": "使用 logging 模块",
            },
            {
                "pattern": "time.sleep(",
                "severity": Severity.MAJOR,
                "category": Category.PERFORMANCE,
                "message": "同步 sleep 会阻塞事件循环",
                "suggestion": "在异步代码中使用 asyncio.sleep()",
            },
        ]


# 演示
def demo_code_review():
    agent = CodeReviewAgent()

    # 模拟一个 PR diff
    diff = FileDiff(
        file_path="src/auth.py",
        added_lines=[
            (15, "    password = 'admin123'"),
            (22, "    result = eval(user_input)"),
            (30, "    print('debug: user logged in')"),
            (45, "except:"),
            (46, "    pass"),
        ],
        removed_lines=[],
    )

    report = agent.review_diff("feat: add user authentication", [diff])

    print(f"  PR: {report.pr_title}")
    print(f"  审查文件: {report.files_reviewed}")
    print(f"  摘要: {report.summary}")
    print(f"  通过: {report.approved}")
    print(f"\n  审查意见:")
    for c in report.comments:
        print(f"    [{c.severity.value}] L{c.line_number} {c.category.value}: {c.message}")
        if c.suggestion:
            print(f"      建议: {c.suggestion}")
```

## 工程化版本

### 上下文感知审查

```python
@dataclass
class FileContext:
    """文件上下文：帮助理解变更的含义。"""
    file_path: str
    full_content: str
    imports: list[str]
    class_names: list[str]
    function_names: list[str]
    related_files: list[str]  # 导入的其他文件


class ContextAwareReviewer:
    """上下文感知审查器。"""

    def __init__(self):
        self.base_agent = CodeReviewAgent()

    def review_with_context(self, diff: FileDiff, context: FileContext) -> list[ReviewComment]:
        """带上下文的审查。"""
        comments = []

        # 基础规则检查
        comments.extend(self.base_agent._review_file(diff))

        # 上下文检查
        comments.extend(self._check_naming_consistency(diff, context))
        comments.extend(self._check_missing_error_handling(diff, context))
        comments.extend(self._check_import_usage(diff, context))

        return comments

    def _check_naming_consistency(self, diff: FileDiff, context: FileContext) -> list[ReviewComment]:
        """检查命名一致性。"""
        comments = []
        for line_no, line in diff.added_lines:
            # 检查函数命名风格
            func_match = re.search(r"def\s+(\w+)", line)
            if func_match:
                func_name = func_match.group(1)
                # 检查是否混用 camelCase 和 snake_case
                if re.search(r"[a-z][A-Z]", func_name):
                    comments.append(ReviewComment(
                        file_path=diff.file_path,
                        line_number=line_no,
                        severity=Severity.MINOR,
                        category=Category.STYLE,
                        message=f"函数名 '{func_name}' 使用 camelCase，项目约定是 snake_case",
                        suggestion=f"改为 '{self._to_snake(func_name)}'",
                        code_snippet=line.strip(),
                    ))
        return comments

    def _check_missing_error_handling(self, diff: FileDiff, context: FileContext) -> list[ReviewComment]:
        """检查缺少错误处理。"""
        comments = []
        risky_patterns = ["open(", "requests.get(", "requests.post(", "db.execute("]

        for line_no, line in diff.added_lines:
            for pattern in risky_patterns:
                if pattern in line:
                    # 检查是否在 try 块中（简化：看前几行）
                    in_try = any(
                        "try:" in prev_line
                        for _, prev_line in diff.added_lines
                        if _ < line_no and _ > line_no - 5
                    )
                    if not in_try:
                        comments.append(ReviewComment(
                            file_path=diff.file_path,
                            line_number=line_no,
                            severity=Severity.MAJOR,
                            category=Category.BUG,
                            message=f"'{pattern.rstrip('(')}' 调用缺少错误处理",
                            suggestion="用 try/except 包裹，处理可能的异常",
                            code_snippet=line.strip(),
                        ))
                    break
        return comments

    def _check_import_usage(self, diff: FileDiff, context: FileContext) -> list[ReviewComment]:
        """检查未使用的导入。"""
        comments = []
        # 简化：检查新增 import 是否在文件中使用
        for line_no, line in diff.added_lines:
            import_match = re.search(r"from\s+\S+\s+import\s+(\w+)", line)
            if import_match:
                name = import_match.group(1)
                # 检查是否在新增行中使用
                used = any(
                    name in other_line
                    for _, other_line in diff.added_lines
                    if _ != line_no
                )
                if not used and name not in context.full_content:
                    comments.append(ReviewComment(
                        file_path=diff.file_path,
                        line_number=line_no,
                        severity=Severity.MINOR,
                        category=Category.STYLE,
                        message=f"导入 '{name}' 似乎未使用",
                        suggestion="如果确实不需要，删除此导入",
                        code_snippet=line.strip(),
                    ))
        return comments

    def _to_snake(self, name: str) -> str:
        return re.sub(r"(?<=[a-z])(?=[A-Z])", "_", name).lower()
```

### 大 PR 分块策略

```python
@dataclass
class ChunkStrategy:
    """大 PR 分块审查策略。"""
    max_lines_per_chunk: int = 200
    max_files_per_chunk: int = 5
    prioritize_security: bool = True


class ChunkedReviewer:
    """分块审查器：处理大 PR。"""

    def __init__(self, strategy: ChunkStrategy | None = None):
        self.strategy = strategy or ChunkStrategy()
        self.reviewer = ContextAwareReviewer()

    def review_large_pr(self, pr_title: str, diffs: list[FileDiff]) -> ReviewReport:
        """分块审查大 PR。"""
        # 按优先级排序文件
        sorted_diffs = self._prioritize_files(diffs)

        # 分块
        chunks = self._split_into_chunks(sorted_diffs)

        # 逐块审查
        all_comments: list[ReviewComment] = []
        for i, chunk in enumerate(chunks):
            chunk_comments = self._review_chunk(chunk)
            all_comments.extend(chunk_comments)

        # 跨文件检查
        cross_file_comments = self._cross_file_check(diffs)
        all_comments.extend(cross_file_comments)

        # 去重
        all_comments = self._deduplicate(all_comments)

        # 生成报告
        severity_order = {Severity.CRITICAL: 0, Severity.MAJOR: 1, Severity.MINOR: 2, Severity.SUGGESTION: 3}
        all_comments.sort(key=lambda c: severity_order[c.severity])

        approved = all(c.severity not in (Severity.CRITICAL, Severity.MAJOR) for c in all_comments)

        return ReviewReport(
            pr_title=pr_title,
            files_reviewed=len(diffs),
            comments=all_comments,
            summary=f"审查 {len(diffs)} 个文件，分 {len(chunks)} 块完成。",
            approved=approved,
        )

    def _prioritize_files(self, diffs: list[FileDiff]) -> list[FileDiff]:
        """按安全优先级排序。"""
        if not self.strategy.prioritize_security:
            return diffs

        security_paths = ["auth", "security", "permission", "token", "crypto"]
        def priority(d: FileDiff) -> int:
            if any(p in d.file_path.lower() for p in security_paths):
                return 0
            return 1

        return sorted(diffs, key=priority)

    def _split_into_chunks(self, diffs: list[FileDiff]) -> list[list[FileDiff]]:
        """分块。"""
        chunks = []
        current_chunk: list[FileDiff] = []
        current_lines = 0

        for diff in diffs:
            file_lines = len(diff.added_lines) + len(diff.removed_lines)
            if (current_lines + file_lines > self.strategy.max_lines_per_chunk
                    or len(current_chunk) >= self.strategy.max_files_per_chunk):
                if current_chunk:
                    chunks.append(current_chunk)
                current_chunk = []
                current_lines = 0
            current_chunk.append(diff)
            current_lines += file_lines

        if current_chunk:
            chunks.append(current_chunk)
        return chunks

    def _review_chunk(self, chunk: list[FileDiff]) -> list[ReviewComment]:
        """审查一个块。"""
        comments = []
        for diff in chunk:
            comments.extend(self.reviewer.base_agent._review_file(diff))
        return comments

    def _cross_file_check(self, diffs: list[FileDiff]) -> list[ReviewComment]:
        """跨文件检查。"""
        comments = []
        # 检查：修改了接口但没修改调用方
        modified_functions = set()
        for diff in diffs:
            for _, line in diff.added_lines:
                match = re.search(r"def\s+(\w+)\(", line)
                if match:
                    modified_functions.add(match.group(1))

        # 简化：实际中需要 AST 分析
        return comments

    def _deduplicate(self, comments: list[ReviewComment]) -> list[ReviewComment]:
        """去重。"""
        seen = set()
        unique = []
        for c in comments:
            key = (c.file_path, c.line_number, c.message)
            if key not in seen:
                seen.add(key)
                unique.append(c)
        return unique
```

## 常见失败模式

### 1. 误报过多

```text
原因：
  规则太宽泛（如：所有 print 都报警）

后果：
  开发者忽略所有审查意见（"狼来了"）

解决：
  1. 区分测试代码和生产代码
  2. 允许白名单（# noqa: review）
  3. 根据文件路径调整规则
  4. 追踪误报率，持续优化
```

### 2. 缺少上下文

```text
原因：
  只看 diff，不看完整文件

后果：
  报告"变量未定义"（实际在上面定义了）

解决：
  1. 读取完整文件内容
  2. 读取相关文件（imports）
  3. 提供足够的上下文给 LLM
```

### 3. 审查意见不可操作

```text
原因：
  "这里写得不好" → 开发者不知道怎么改

后果：
  意见被忽略

解决：
  1. 每条意见必须带具体建议
  2. 给出修改后的代码示例
  3. 解释"为什么"要改
```

### 4. 审查太慢

```text
原因：
  大 PR（50+ 文件）逐行 LLM 审查

后果：
  开发者等不了，直接合并

解决：
  1. 规则引擎先过滤（毫秒级）
  2. 只对"可疑"代码调用 LLM
  3. 并行审查多个文件
  4. 设置超时（5 分钟内完成）
```

## 什么时候不要这么做

```text
不需要代码审查 Agent 的场景：

1. 纯配置变更
   改个 YAML / JSON 配置，不需要逻辑审查

2. 自动生成的代码
   protobuf、swagger codegen 的输出
   审查生成器模板更有意义

3. 极小团队（1-2 人）
   自己写自己看，Agent 审查价值有限

4. 已有完善 CI
   lint + type check + test 全覆盖
   Agent 的增量价值小
```

### 配置化规则

```python
@dataclass
class ReviewConfig:
    """审查配置（每个项目可不同）。"""
    enabled_categories: list[Category] = field(default_factory=lambda: list(Category))
    min_severity_to_block: Severity = Severity.CRITICAL
    ignore_patterns: list[str] = field(default_factory=lambda: [
        "test_*", "*_test.py", "conftest.py", "migrations/*"
    ])
    max_comments: int = 20  # 避免评论轰炸
    language: str = "python"
    custom_rules: list[dict] = field(default_factory=list)

    def should_ignore(self, file_path: str) -> bool:
        """检查文件是否应忽略。"""
        import fnmatch
        return any(fnmatch.fnmatch(file_path, p) for p in self.ignore_patterns)

    def should_report(self, comment: ReviewComment) -> bool:
        """检查是否应报告此意见。"""
        if comment.category not in self.enabled_categories:
            return False
        return True
```

## 如何观测和评测

### 评测指标

```text
准确性指标：
  - 精确率（Precision）：报告的问题中，真正是问题的比例
    目标：> 80%（误报率 < 20%）
  - 召回率（Recall）：实际问题中，被发现的比例
    目标：> 60%（不要求发现所有问题）

效率指标：
  - 审查耗时：< 3 分钟（中等 PR）
  - 人工审查时间减少：> 30%

质量指标：
  - 开发者采纳率：> 50%
  - 误报标记率：< 15%
  - 漏报率（上线后发现的 bug）：持续下降
```

## 面试怎么讲

```text
问：你怎么设计一个代码审查 Agent？

答（30 秒版）：
  两层架构：规则引擎 + LLM。
  规则引擎做快速静态检查（安全、风格），毫秒级。
  LLM 做深度审查（逻辑、设计），只对可疑代码调用。
  大 PR 分块处理，安全文件优先。
  每条意见带严重级别、分类、具体建议。
  只有 Critical 才阻塞合并，避免过度干预。

追问 1：怎么减少误报？
  → 白名单机制（# noqa）、路径过滤（测试文件跳过）、
    追踪误报率持续调优、允许开发者标记"不是问题"。

追问 2：大 PR 怎么处理？
  → 分块：每块最多 200 行 / 5 个文件。
    优先级：安全相关文件先审查。
    跨文件检查：接口变更是否同步了调用方。

追问 3：和 linter 有什么区别？
  → Linter 检查语法和风格（确定性规则）。
    Agent 理解语义和逻辑（"这个 if 条件可能漏了边界"）。
    Agent 能给设计建议，linter 不能。
```

```python
class CommentFormatter:
    """审查意见格式化器（适配不同平台）。"""

    def to_github_inline(self, comment: ReviewComment) -> dict:
        """GitHub inline comment 格式。"""
        body = f"**[{comment.severity.value.upper()}]** {comment.message}\n\n"
        if comment.suggestion:
            body += f"**建议：** {comment.suggestion}\n\n"
        if comment.code_snippet:
            body += f"```\n{comment.code_snippet}\n```\n"
        body += f"\n_分类: {comment.category.value}_"

        return {
            "path": comment.file_path,
            "line": comment.line_number,
            "body": body,
        }

    def to_markdown_summary(self, report: ReviewReport) -> str:
        """生成 Markdown 摘要评论。"""
        lines = [
            f"## 🤖 Code Review Summary",
            "",
            f"**结果:** {'✅ 通过' if report.approved else '❌ 需要修改'}",
            "",
            f"| 严重度 | 数量 |",
            f"|--------|------|",
            f"| 🔴 Critical | {report.critical_count} |",
            f"| 🟠 Major | {report.major_count} |",
            f"| 🟡 Minor | {sum(1 for c in report.comments if c.severity == Severity.MINOR)} |",
            f"| 💡 Suggestion | {sum(1 for c in report.comments if c.severity == Severity.SUGGESTION)} |",
            "",
            "### 详细意见",
            "",
        ]

        for c in report.comments[:10]:
            lines.append(f"- **{c.file_path}:{c.line_number}** [{c.severity.value}] {c.message}")

        if len(report.comments) > 10:
            lines.append(f"- ... 及其他 {len(report.comments) - 10} 条")

        return "\n".join(lines)

    def to_json(self, report: ReviewReport) -> dict:
        """生成 JSON（用于 API）。"""
        return {
            "pr_title": report.pr_title,
            "approved": report.approved,
            "summary": report.summary,
            "files_reviewed": report.files_reviewed,
            "comments": [
                {
                    "file": c.file_path,
                    "line": c.line_number,
                    "severity": c.severity.value,
                    "category": c.category.value,
                    "message": c.message,
                    "suggestion": c.suggestion,
                }
                for c in report.comments
            ],
        }
```

### 自定义规则引擎

```python
class CustomRuleEngine:
    """可扩展的规则引擎。"""

    def __init__(self):
        self._rules: list[dict] = []

    def add_rule(self, name: str, pattern: str, severity: Severity,
                 category: Category, message: str, suggestion: str = "",
                 file_pattern: str = "*"):
        """添加自定义规则。"""
        self._rules.append({
            "name": name,
            "pattern": pattern,
            "severity": severity,
            "category": category,
            "message": message,
            "suggestion": suggestion,
            "file_pattern": file_pattern,
        })

    def check(self, diff: FileDiff) -> list[ReviewComment]:
        """执行规则检查。"""
        import fnmatch
        comments = []

        for rule in self._rules:
            if not fnmatch.fnmatch(diff.file_path, rule["file_pattern"]):
                continue

            for line_no, line in diff.added_lines:
                if rule["pattern"] in line:
                    comments.append(ReviewComment(
                        file_path=diff.file_path,
                        line_number=line_no,
                        severity=rule["severity"],
                        category=rule["category"],
                        message=rule["message"],
                        suggestion=rule["suggestion"],
                        code_snippet=line.strip(),
                    ))

        return comments


# 使用示例
def setup_custom_rules() -> CustomRuleEngine:
```
