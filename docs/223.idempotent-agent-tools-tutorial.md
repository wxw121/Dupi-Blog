## 你会学到什么

- 理解幂等性在 Agent 工具中的重要性。
- 用 idempotency key 实现请求去重。
- 设计"先查后写"的安全模式。
- 实现乐观锁和状态机防重。
- 让 Agent 的重试不会造成副作用。

## 它解决什么问题

```text
Agent 为什么需要幂等工具？

场景 1：超时重试
  Agent 调用 create_order → 超时
  Agent 重试 → 又创建了一个订单
  结果：用户被扣了两次钱

场景 2：LLM 重复调用
  LLM 不确定上次是否成功
  再次调用 send_email
  结果：用户收到两封相同邮件

场景 3：并发执行
  两个 Agent 实例同时处理同一任务
  都调用 deduct_inventory
  结果：库存多扣了一次

场景 4：断点恢复
  Agent 从 checkpoint 恢复
  重新执行已完成的步骤
  结果：操作重复执行

核心问题：
  分布式系统中，"恰好一次"执行是不可能的
  只能做到"至少一次"执行 + 幂等 = "恰好一次"效果
```

## 幂等的三种实现模式

```text
模式 1：天然幂等
  操作本身重复执行无副作用
  例：GET 查询、SET 覆盖写、UPSERT
  适用：读操作、状态设置

模式 2：Idempotency Key（幂等键）
  每次操作带唯一标识
  服务端记录已处理的 key
  重复 key 直接返回上次结果
  适用：创建类操作（POST）

模式 3：状态机 + 条件写
  操作前检查当前状态
  只在预期状态下执行
  例：订单只能从 "pending" → "paid"
  适用：状态转换类操作

选择指南：
  读操作 → 天然幂等，不需要额外设计
  创建操作 → Idempotency Key
  状态转换 → 状态机 + 条件写
  删除操作 → 软删除 + 状态检查
```

## 最小示例

### 非幂等 vs 幂等

```python
# ❌ 非幂等：每次调用都创建新订单
class NonIdempotentOrderTool:
    def __init__(self):
        self.orders = []
        self._counter = 0

    def create_order(self, item: str, quantity: int) -> dict:
        self._counter += 1
        order = {"id": f"ORD-{self._counter}", "item": item, "qty": quantity}
        self.orders.append(order)
        return order

    # 调用两次 → 两个订单！


# ✓ 幂等：相同 idempotency_key 只创建一次
class IdempotentOrderTool:
    def __init__(self):
        self.orders = []
        self._counter = 0
        self._processed_keys: dict[str, dict] = {}

    def create_order(
        self, item: str, quantity: int, idempotency_key: str
    ) -> dict:
        # 检查是否已处理
        if idempotency_key in self._processed_keys:
            return self._processed_keys[idempotency_key]

        # 首次执行
        self._counter += 1
        order = {"id": f"ORD-{self._counter}", "item": item, "qty": quantity}
        self.orders.append(order)

        # 记录已处理
        self._processed_keys[idempotency_key] = order
        return order

    # 调用两次（相同 key）→ 只有一个订单！
```

### 状态机防重

```python
from enum import Enum


class OrderState(str, Enum):
    PENDING = "pending"
    PAID = "paid"
    SHIPPED = "shipped"
    CANCELLED = "cancelled"


# 合法的状态转换
VALID_TRANSITIONS = {
    OrderState.PENDING: {OrderState.PAID, OrderState.CANCELLED},
    OrderState.PAID: {OrderState.SHIPPED, OrderState.CANCELLED},
    OrderState.SHIPPED: set(),  # 终态
    OrderState.CANCELLED: set(),  # 终态
}


class OrderStateMachine:
    """订单状态机：防止非法状态转换"""

    def __init__(self):
        self._orders: dict[str, OrderState] = {}

    def create(self, order_id: str):
        self._orders[order_id] = OrderState.PENDING

    def transition(self, order_id: str, target: OrderState) -> bool:
        """尝试状态转换（幂等）"""
        current = self._orders.get(order_id)
        if current is None:
            raise ValueError(f"Order {order_id} not found")

        # 已经是目标状态 → 幂等返回成功
        if current == target:
            return True

        # 检查转换是否合法
        if target not in VALID_TRANSITIONS[current]:
            raise ValueError(
                f"Cannot transition from {current.value} to {target.value}"
            )

        self._orders[order_id] = target
        return True

    def get_state(self, order_id: str) -> OrderState:
        return self._orders[order_id]
```

## 工程化版本

### Idempotency Key 管理器

```python
import hashlib
import json
import time
from dataclasses import dataclass, field
from typing import Any


@dataclass
class IdempotencyRecord:
    """幂等记录"""
    key: str
    result: Any
    created_at: float
    expires_at: float
    status: str = "completed"  # processing / completed / failed


class IdempotencyStore:
    """幂等键存储（生产环境用 Redis）"""

    def __init__(self, ttl_seconds: int = 86400):
        self._store: dict[str, IdempotencyRecord] = {}
        self.ttl = ttl_seconds

    def generate_key(self, tool_name: str, params: dict) -> str:
        """从工具名和参数生成幂等键"""
        raw = f"{tool_name}:{json.dumps(params, sort_keys=True)}"
        return hashlib.sha256(raw.encode()).hexdigest()[:32]

    def check(self, key: str) -> IdempotencyRecord | None:
        """检查是否已处理"""
        record = self._store.get(key)
        if record is None:
            return None
        # 检查过期
        if time.time() > record.expires_at:
            del self._store[key]
            return None
        return record

    def mark_processing(self, key: str):
        """标记为处理中（防并发）"""
        self._store[key] = IdempotencyRecord(
            key=key,
            result=None,
            created_at=time.time(),
            expires_at=time.time() + self.ttl,
            status="processing",
        )

    def mark_completed(self, key: str, result: Any):
        """标记为已完成"""
        if key in self._store:
            self._store[key].result = result
            self._store[key].status = "completed"

    def mark_failed(self, key: str):
        """标记为失败（允许重试）"""
        if key in self._store:
            del self._store[key]

    def cleanup_expired(self):
        """清理过期记录"""
        now = time.time()
        expired = [k for k, v in self._store.items() if now > v.expires_at]
        for k in expired:
            del self._store[k]
```

### 幂等工具装饰器

```python
from functools import wraps
from typing import Callable


def idempotent(store: IdempotencyStore):
    """幂等装饰器"""
    def decorator(func: Callable):
        @wraps(func)
        async def wrapper(*args, **kwargs):
            # 生成幂等键
            tool_name = func.__name__
            params = kwargs or (args[1:] if args else {})
            key = store.generate_key(tool_name, params if isinstance(params, dict) else {})

            # 检查是否已处理
            record = store.check(key)
            if record:
                if record.status == "completed":
                    return record.result  # 直接返回上次结果
                elif record.status == "processing":
                    raise RuntimeError(
                        f"Operation '{tool_name}' is already in progress"
                    )

            # 标记处理中
            store.mark_processing(key)

            try:
                result = await func(*args, **kwargs)
                store.mark_completed(key, result)
                return result
            except Exception:
                store.mark_failed(key)
                raise

        return wrapper
    return decorator


# 使用示例
idem_store = IdempotencyStore(ttl_seconds=3600)


class PaymentService:
    """支付服务"""

    @idempotent(idem_store)
    async def charge(self, user_id: str, amount: float, order_id: str) -> dict:
        """扣款（幂等）"""
        # 实际扣款逻辑
        return {
            "transaction_id": f"TXN-{order_id}",
            "user_id": user_id,
            "amount": amount,
            "status": "charged",
        }

    @idempotent(idem_store)
    async def send_notification(self, user_id: str, message: str) -> dict:
        """发送通知（幂等）"""
        return {"sent_to": user_id, "message": message, "status": "sent"}
```

### 条件写入器

```python
class ConditionalWriter:
    """条件写入：只在满足前置条件时执行"""

    def __init__(self):
        self._resources: dict[str, dict] = {}

    def create_if_not_exists(self, resource_id: str, data: dict) -> tuple[dict, bool]:
        """不存在则创建（幂等）"""
        if resource_id in self._resources:
            return self._resources[resource_id], False  # 已存在，未创建
        self._resources[resource_id] = data
        return data, True  # 新创建

    def update_if_version(self, resource_id: str, data: dict,
                          expected_version: int) -> tuple[dict, bool]:
        """乐观锁更新（版本匹配才写入）"""
        current = self._resources.get(resource_id)
        if current is None:
            raise ValueError(f"Resource {resource_id} not found")

        current_version = current.get("_version", 0)
        if current_version != expected_version:
            raise ValueError(
                f"Version conflict: expected {expected_version}, "
                f"got {current_version}"
            )

        data["_version"] = current_version + 1
        self._resources[resource_id] = data
        return data, True

    def delete_if_exists(self, resource_id: str) -> bool:
        """存在则删除（幂等）"""
        if resource_id in self._resources:
            del self._resources[resource_id]
            return True
        return False  # 已不存在，幂等成功
```

### 完整幂等执行器

```python
class IdempotentToolExecutor:
    """幂等工具执行器"""

    def __init__(self):
        self._tools: dict[str, Callable] = {}
        self._idempotency_store = IdempotencyStore()
        self._state_machines: dict[str, Any] = {}
        self._conditional_writer = ConditionalWriter()

    def register(self, name: str, handler: Callable, idempotent: bool = True):
        self._tools[name] = handler
        if not idempotent:
            # 非幂等工具不自动重试
            pass

    async def execute(
        self, tool_name: str, params: dict, idempotency_key: str | None = None
    ) -> dict:
        """执行工具（幂等保证）"""
        if tool_name not in self._tools:
            return {"success": False, "error": f"Tool '{tool_name}' not found"}

        # 生成或使用提供的幂等键
        key = idempotency_key or self._idempotency_store.generate_key(
            tool_name, params
        )

        # 检查幂等记录
        record = self._idempotency_store.check(key)
        if record and record.status == "completed":
            return {
                "success": True,
                "data": record.result,
                "idempotent_replay": True,
            }

        # 执行
        self._idempotency_store.mark_processing(key)
        try:
            handler = self._tools[tool_name]
            result = await handler(**params)
            self._idempotency_store.mark_completed(key, result)
            return {"success": True, "data": result, "idempotent_replay": False}
        except Exception as e:
            self._idempotency_store.mark_failed(key)
            return {"success": False, "error": str(e)}
```

## 常见失败模式

### 1. 幂等键冲突

```text
症状：
  不同操作被误判为重复

原因：
  幂等键生成规则太粗糙
  例：只用 tool_name 做 key

解决：
  key = hash(tool_name + sorted_params + timestamp_window)
  确保不同参数生成不同 key
  时间窗口内相同参数才算重复
```

### 2. 幂等记录过期

```text
症状：
  过期后重试，操作重复执行

原因：
  TTL 太短（如 5 分钟）
  但 Agent 可能 10 分钟后才重试

解决：
  写操作的幂等记录 TTL ≥ 24 小时
  关键操作（支付）永久保留
  过期前告警
```

### 3. 并发竞争

```text
症状：
  两个请求同时通过"不存在"检查
  都执行了创建

原因：
  check-then-act 不是原子的

解决：
  用分布式锁（Redis SETNX）
  或数据库唯一约束
  或 CAS（Compare-And-Swap）
```

### 4. 部分成功

```text
症状：
  批量操作中部分成功部分失败
  重试时成功的部分重复执行

原因：
  批量操作没有细粒度幂等

解决：
  每个子操作有独立幂等键
  记录每个子操作的状态
  重试时跳过已成功的
```

## 什么时候不要这么做

```text
不需要幂等设计的场景：

1. 纯读操作
   → GET / SELECT 天然幂等
   → 不需要额外设计

2. 覆盖写操作
   → SET / UPDATE 到固定值
   → 重复执行结果相同

3. 有唯一约束的创建
   → 数据库 UNIQUE 约束保证不重复
   → 重复插入会报错（可捕获）

4. 一次性脚本
   → 不会重试
   → 不需要幂等
```

## 生产环境注意事项

### 分布式幂等（Redis 实现）

```python
class RedisIdempotencyStore:
    """基于 Redis 的分布式幂等存储"""

    def __init__(self, redis_client, ttl: int = 86400):
        self.redis = redis_client
        self.ttl = ttl
        self.prefix = "idem:"

    async def try_acquire(self, key: str) -> bool:
        """尝试获取幂等锁（原子操作）"""
        full_key = f"{self.prefix}{key}"
        # SETNX: 只在 key 不存在时设置
        acquired = await self.redis.set(
            full_key,
            "processing",
            nx=True,  # 不存在才设置
            ex=self.ttl,  # 过期时间
        )
        return acquired is not None

    async def get_result(self, key: str) -> Any | None:
        """获取已完成的结果"""
        full_key = f"{self.prefix}{key}:result"
        data = await self.redis.get(full_key)
        if data:
            return json.loads(data)
        return None

    async def store_result(self, key: str, result: Any):
        """存储完成结果"""
        full_key = f"{self.prefix}{key}:result"
        await self.redis.set(
            full_key,
            json.dumps(result),
            ex=self.ttl,
        )
        # 更新状态
        await self.redis.set(
            f"{self.prefix}{key}",
            "completed",
            ex=self.ttl,
        )

    async def release_on_failure(self, key: str):
        """失败时释放锁（允许重试）"""
        await self.redis.delete(f"{self.prefix}{key}")
```

### 幂等键的最佳实践

```text
幂等键设计原则：

1. 由调用方生成
   → Agent 在发起调用时生成 key
   → 重试时使用相同的 key
   → 格式：{tool_name}:{uuid4}

2. 包含业务语义
   → 支付：payment:{order_id}:{amount}
   → 发货：ship:{order_id}:{tracking_no}
   → 通知：notify:{user_id}:{template}:{date}

3. 时间窗口
   → 相同参数 + 相同时间窗口 = 相同 key
   → 窗口大小取决于业务（通常 1-24 小时）

4. 不可预测
   → 不要用自增 ID
   → 用 UUID 或 hash
   → 防止外部伪造
```

### 数据库层幂等

```python
class DatabaseIdempotency:
    """数据库层面的幂等保证"""

    CREATE_TABLE_SQL = """
    CREATE TABLE IF NOT EXISTS idempotency_keys (
        key VARCHAR(64) PRIMARY KEY,
        tool_name VARCHAR(100) NOT NULL,
        status VARCHAR(20) NOT NULL DEFAULT 'processing',
        result JSONB,
        created_at TIMESTAMP DEFAULT NOW(),
        expires_at TIMESTAMP NOT NULL
    );
    CREATE INDEX idx_idem_expires ON idempotency_keys(expires_at);
    """

    INSERT_SQL = """
    INSERT INTO idempotency_keys (key, tool_name, status, expires_at)
    VALUES (%s, %s, 'processing', NOW() + INTERVAL '%s seconds')
    ON CONFLICT (key) DO NOTHING;
    """

    CHECK_SQL = """
    SELECT status, result FROM idempotency_keys
    WHERE key = %s AND expires_at > NOW();
    """

    COMPLETE_SQL = """
    UPDATE idempotency_keys
    SET status = 'completed', result = %s
    WHERE key = %s;
    """

    # 利用 PRIMARY KEY 约束保证原子性
    # INSERT ... ON CONFLICT DO NOTHING 是原子的
    # 不需要额外的分布式锁
```

## 如何观测和评测

### 幂等指标

```text
关键监控指标：

1. 幂等命中率
   - idempotent_replays / total_calls
   - 正常 < 5%
   - > 20% 说明重试太多，排查根因

2. 并发冲突率
   - conflicts / total_calls
   - 正常 < 1%
   - > 5% 说明有并发问题

3. 幂等键存储量
   - 当前活跃 key 数量
   - 增长趋势
   - 及时清理过期 key

4. 部分成功率
   - 批量操作中子操作的成功/失败比
   - 重试后是否全部成功
```

## 和 RAG / 后端 / 前端的关系

```text
幂等设计在系统中的位置：

RAG 管道：
  文档索引是幂等的（相同文档重复索引 = 覆盖）
  用 doc_hash 做幂等键
  重复索引不会产生重复 chunk

后端 API：
  REST API 的 PUT/DELETE 天然幂等
  POST 需要 Idempotency-Key header
  数据库用 UNIQUE 约束兜底

前端展示：
  前端按钮防重复点击
  请求带 idempotency_key
  重复响应不重复渲染
```

```text
 - 并发用分布式锁或数据库约束

 实际效果：
 重试导致的重复操作降为零。"

追问 1：幂等键怎么生成？
  → tool_name + params_hash + time_window。
    或由 Agent 显式传入 UUID。
    关键操作用业务 ID（如 order_id）。

追问 2：Redis 挂了怎么办？
  → 降级为数据库唯一约束。
    INSERT ON CONFLICT DO NOTHING。
    性能降低但正确性保证。

追问 3：批量操作怎么幂等？
  → 每个子操作独立幂等键。
    记录每个子操作状态。
    重试时跳过已成功的。
```

## 实战案例：电商订单系统的幂等设计

### 完整订单工具集

```python
class EcommerceIdempotentTools:
    """电商幂等工具集"""

    def __init__(self):
        self.idem_store = IdempotencyStore(ttl_seconds=86400)
        self.order_sm = OrderStateMachine()
        self.writer = ConditionalWriter()

    async def create_order(
        self, user_id: str, items: list[dict], idempotency_key: str
    ) -> dict:
        """创建订单（幂等）"""
        # 检查幂等键
        record = self.idem_store.check(idempotency_key)
        if record and record.status == "completed":
            return {**record.result, "_replayed": True}

        self.idem_store.mark_processing(idempotency_key)
        try:
            # 创建订单
            order_id = f"ORD-{hashlib.md5(idempotency_key.encode()).hexdigest()[:8].upper()}"
            order = {
                "order_id": order_id,
                "user_id": user_id,
                "items": items,
                "status": "pending",
                "created_at": time.time(),
            }
            self.order_sm.create(order_id)
            self.writer.create_if_not_exists(order_id, order)

            self.idem_store.mark_completed(idempotency_key, order)
            return order
        except Exception:
            self.idem_store.mark_failed(idempotency_key)
            raise

    async def pay_order(
        self, order_id: str, amount: float, idempotency_key: str
    ) -> dict:
        """支付订单（幂等 + 状态机）"""
        # 幂等检查
        record = self.idem_store.check(idempotency_key)
        if record and record.status == "completed":
            return {**record.result, "_replayed": True}

        # 状态机检查
        current_state = self.order_sm.get_state(order_id)
        if current_state == OrderState.PAID:
            # 已支付，幂等返回
            return {"order_id": order_id, "status": "paid", "_replayed": True}

        if current_state != OrderState.PENDING:
            raise ValueError(
                f"Order {order_id} in state '{current_state.value}', "
                f"cannot pay"
            )

        self.idem_store.mark_processing(idempotency_key)
        try:
            # 执行支付
            self.order_sm.transition(order_id, OrderState.PAID)
            result = {
                "order_id": order_id,
                "amount": amount,
                "status": "paid",
                "transaction_id": f"TXN-{idempotency_key[:12]}",
            }
            self.idem_store.mark_completed(idempotency_key, result)
            return result
        except Exception:
            self.idem_store.mark_failed(idempotency_key)
            raise

    async def cancel_order(
        self, order_id: str, reason: str, idempotency_key: str
    ) -> dict:
        """取消订单（幂等 + 状态机）"""
        record = self.idem_store.check(idempotency_key)
        if record and record.status == "completed":
            return {**record.result, "_replayed": True}

        current_state = self.order_sm.get_state(order_id)
        if current_state == OrderState.CANCELLED:
            return {"order_id": order_id, "status": "cancelled", "_replayed": True}

        if current_state not in (OrderState.PENDING, OrderState.PAID):
            raise ValueError(
                f"Order {order_id} in state '{current_state.value}', "
                f"cannot cancel"
            )

        self.idem_store.mark_processing(idempotency_key)
        try:
            self.order_sm.transition(order_id, OrderState.CANCELLED)
            result = {
                "order_id": order_id,
                "status": "cancelled",
                "reason": reason,
            }
            self.idem_store.mark_completed(idempotency_key, result)
            return result
        except Exception:
            self.idem_store.mark_failed(idempotency_key)
            raise
```

### 幂等设计检查清单

```text
上线前幂等性检查：

□ 所有写操作都有幂等键参数
□ 幂等键由调用方（Agent）生成
□ 幂等记录 TTL ≥ 24 小时
□ 支付类操作幂等记录永久保留
□ 状态转换用状态机保护
□ 并发场景用分布式锁或 DB 约束
□ 批量操作每个子项独立幂等
□ 重试时复用相同幂等键
□ 幂等命中时返回原始结果
□ 日志记录幂等命中事件
□ 监控幂等命中率指标
□ 定期清理过期幂等记录
```

## 设计原则总结

```text
幂等设计 6 条原则：

1. 每个写操作都有幂等键
   由调用方生成，重试时复用
   格式：{action}:{resource}:{identifier}

2. 先查后写
   执行前检查是否已完成
   已完成则直接返回上次结果

3. 状态机保护状态转换
   只允许合法转换
   已在目标状态则幂等成功

4. 原子性保证
   用 SETNX / UNIQUE / CAS
   防止并发竞争

5. 幂等记录有生命周期
   普通操作 24h
   金融操作永久
   定期清理过期记录

6. 透明性
   幂等命中时标记 _replayed
   日志记录幂等事件
   监控幂等命中率
```

---

> **核心要点：** 幂等不是“不做”，而是“做了多次和做一次效果一样”。这是分布式系统的基石，也是 Agent 可靠性的保障。
